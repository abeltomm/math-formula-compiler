# Valid formulas.  Format:  formula | expected AST (prefix / S-expression form)
# "\n" inside a formula is replaced by a real newline by the test runner.
2 + 3                     | (+ 2 3)
2 + 3 * 4                 | (+ 2 (* 3 4))
(a + b) * c               | (* (+ a b) c)
x^2 + 2*x + 1             | (+ (+ (^ x 2) (* 2 x)) 1)
sqrt(25)                  | (call sqrt 25)
sqrt(x^2 + y^2)           | (call sqrt (+ (^ x 2) (^ y 2)))
pow(2, 3)                 | (call pow 2 3)
pow(a + b, 2)             | (call pow (+ a b) 2)
-x + 4.5                  | (+ (neg x) 4.5)
2^3^2                     | (^ 2 (^ 3 2))
-2^2                      | (neg (^ 2 2))
2^-3                      | (^ 2 (neg 3))
10 - 4 - 3                | (- (- 10 4) 3)
8 / 4 / 2                 | (/ (/ 8 4) 2)
pow(2, 3) + 5             | (+ (call pow 2 3) 5)
+x                        | (pos x)
((1))                     | 1
2 * (3 + 4)^2             | (* 2 (^ (+ 3 4) 2))
sqrt(pow(3,2) + pow(4,2)) | (call sqrt (+ (call pow 3 2) (call pow 4 2)))
2 +\n3 * radius           | (+ 2 (* 3 radius))
